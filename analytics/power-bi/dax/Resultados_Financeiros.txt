FORMULAS DAX FINAIS — RESULTADOS FINANCEIROS (PÁGINA 2)

TABELA BASE
'dataset_01_agendamentos_conversao_200k'

1. Receita_Potencial
Receita_Potencial =
SUM('dataset_01_agendamentos_conversao_200k'[ticket_estimado])

2. Receita_Convertida
Receita_Convertida =
CALCULATE(
    SUM('dataset_01_agendamentos_conversao_200k'[ticket_estimado]),
    'dataset_01_agendamentos_conversao_200k'[target_conversao] = 1
)

3. Gap_Receita
Gap_Receita =
[Receita_Potencial] - [Receita_Convertida]

4. Aproveitamento_Receita
Aproveitamento_Receita =
DIVIDE([Receita_Convertida], [Receita_Potencial], 0)

5. Ticket_Medio
Ticket_Medio =
DIVIDE([Receita_Potencial], [Qtd_Agendamentos], 0)

6. Aproveitamento_Receita_Especialidade
Aproveitamento_Receita_Especialidade =
DIVIDE([Receita_Convertida], [Receita_Potencial], 0)
