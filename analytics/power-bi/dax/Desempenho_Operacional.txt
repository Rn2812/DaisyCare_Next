FORMULAS DAX FINAIS — DESEMPENHO OPERACIONAL (PÁGINA 1)

TABELA BASE
'dataset_01_agendamentos_conversao_200k'

1. Qtd_Agendamentos
Qtd_Agendamentos =
COUNT('dataset_01_agendamentos_conversao_200k'[id_agendamento])

2. Qtd_Convertidos
Qtd_Convertidos =
SUM('dataset_01_agendamentos_conversao_200k'[target_conversao])

3. Qtd_NoShow
Qtd_NoShow =
SUM('dataset_01_agendamentos_conversao_200k'[no_show])

4. Qtd_Cancelados
Qtd_Cancelados =
SUM('dataset_01_agendamentos_conversao_200k'[cancelado])

5. Qtd_Remarcados
Qtd_Remarcados =
SUM('dataset_01_agendamentos_conversao_200k'[remarcado])

6. Taxa_Conversao
Taxa_Conversao =
DIVIDE([Qtd_Convertidos], [Qtd_Agendamentos], 0)

7. Taxa_NoShow
Taxa_NoShow =
DIVIDE([Qtd_NoShow], [Qtd_Agendamentos], 0)

8. Taxa_Cancelamento
Taxa_Cancelamento =
DIVIDE([Qtd_Cancelados], [Qtd_Agendamentos], 0)

9. Impacto_NoShow
Impacto_NoShow =
CALCULATE(
    SUM('dataset_01_agendamentos_conversao_200k'[ticket_estimado]),
    'dataset_01_agendamentos_conversao_200k'[no_show] = 1
)

10. Perda_Financeira_Especialidade
Perda_Financeira_Especialidade =
CALCULATE(
    SUM('dataset_01_agendamentos_conversao_200k'[ticket_estimado]),
    'dataset_01_agendamentos_conversao_200k'[no_show] = 1
)
