    'dataset_03_logs_daisy_n8n_IA_logistica_200k'[evento] = "vaga_liberada"
)

2. Qtd_Pacientes_Aptos
Qtd_Pacientes_Aptos =
CALCULATE(
    COUNTROWS('dataset_03_logs_daisy_n8n_IA_logistica_200k'),
    'dataset_03_logs_daisy_n8n_IA_logistica_200k'[paciente_apto] = 1
)

3. Qtd_Convites_Enviados
Qtd_Convites_Enviados =
CALCULATE(
    COUNTROWS('dataset_03_logs_daisy_n8n_IA_logistica_200k'),
    'dataset_03_logs_daisy_n8n_IA_logistica_200k'[convite_enviado_flag] = 1
)

4. Qtd_Convites_Aceitos
Qtd_Convites_Aceitos =
CALCULATE(
    COUNTROWS('dataset_03_logs_daisy_n8n_IA_logistica_200k'),
    'dataset_03_logs_daisy_n8n_IA_logistica_200k'[convite_aceito_flag] = 1
)

5. Qtd_Encaixes_Confirmados
Qtd_Encaixes_Confirmados =
CALCULATE(
    COUNTROWS('dataset_03_logs_daisy_n8n_IA_logistica_200k'),
    'dataset_03_logs_daisy_n8n_IA_logistica_200k'[encaixe_confirmado_flag] = 1
)

6. Taxa_Aceite
Taxa_Aceite =
DIVIDE([Qtd_Convites_Aceitos], [Qtd_Convites_Enviados], 0)

7. Taxa_Aproveitamento
Taxa_Aproveitamento =
DIVIDE([Qtd_Encaixes_Confirmados], [Qtd_Vagas_Liberadas], 0)

8. Score_Medio
Score_Medio =
AVERAGE('dataset_03_logs_daisy_n8n_IA_logistica_200k'[score_viabilidade])

9. Receita_Em_Risco_IA
Receita_Em_Risco_IA =
VAR AgendamentosRisco =
    CALCULATETABLE(
        VALUES('dataset_03_logs_daisy_n8n_IA_logistica_200k'[id_agendamento]),
        'dataset_03_logs_daisy_n8n_IA_logistica_200k'[evento] = "vaga_liberada"
    )
RETURN
CALCULATE(
    SUM('dataset_01_agendamentos_conversao_200k'[ticket_estimado]),
    TREATAS(
        AgendamentosRisco,
        'dataset_01_agendamentos_conversao_200k'[id_agendamento]
    )
)

10. Receita_Preservada_IA
Receita_Preservada_IA =
VAR AgendamentosPreservados =
    CALCULATETABLE(
        VALUES('dataset_03_logs_daisy_n8n_IA_logistica_200k'[id_agendamento]),
        'dataset_03_logs_daisy_n8n_IA_logistica_200k'[encaixe_confirmado_flag] = 1
    )
RETURN
CALCULATE(
    SUM('dataset_01_agendamentos_conversao_200k'[ticket_estimado]),
    TREATAS(
        AgendamentosPreservados,
        'dataset_01_agendamentos_conversao_200k'[id_agendamento]
    )
)

11. Taxa_Preservacao_Receita_IA
Taxa_Preservacao_Receita_IA =
DIVIDE([Receita_Preservada_IA], [Receita_Em_Risco_IA], 0)
